cmake_minimum_required(VERSION 3.22)
project(PandoraReconstructed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The confirmed syscall numbers and layouts require Android arm64-v8a")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../pandora-core"
                 "${CMAKE_CURRENT_BINARY_DIR}/pandora-core")
add_library(PandoraReconstructed SHARED Main.cpp JniBridge.cpp)
set_target_properties(PandoraReconstructed PROPERTIES OUTPUT_NAME "pandora-core")
target_compile_options(PandoraReconstructed PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(PandoraReconstructed PRIVATE pandora-core log)
